CD C:\Program files\Qemu


Create HDD Image------


IMG Format>

qemu-img create -f qcow2 (harddiskimagelocation).img 10G


Enter Setup----------------


qemu-system-ppc -L pc-bios -boot d -M mac99 -m 2048 -hda (harddiskimagelocation) -cdrom (ISOlocation)


Go TO Disk 2--------------


qemu-system-ppc -L pc-bios -boot c -M mac99 -m 2048 -hda (harddiskimagelocation) -cdrom (ISOlocation)


Start Mac OS--------------


qemu-system-ppc -L pc-bios -boot c -M mac99 -m 2048 -hda (harddiskimagelocation)


Mac OS 9x--------------------------------------------


Enter Setup

qemu-system-ppc -L pc-bios -boot d -M mac99 -m 1024 -hda (harddiskimagelocation) -cdrom (ISOlocation)


Start Mac OS

qemu-system-ppc -L pc-bios -boot c -M mac99 -m 1024 -hda (harddiskimagelocation)


Super ppc for Mac OS X 10.5 Leopard------------


Start setup--------------------------------------------


qemu-system-ppc -L pc-bios -boot d -M mac99,via=pmu -m 2048 -prom-env "auto-boot?=true" -prom-env "boot=args=-V" -prom-env "vga-ndrv?=true" -drive file=(harddiskimagelocation) -drive file=(ISO location),format=raw,media=cdrom -sdl -netdev user,id=network01 -device sungem,netdev=network01 -device VGA,edid=on -g 1366x768x32


Start from Hard disk---------------------------------


qemu-system-ppc -L pc-bios -boot c -M mac99,via=pmu -m 2048 -prom-env "auto-boot?=true" -prom-env "boot=args=-V" -prom-env "vga-ndrv?=true" -drive file=(harddiskimagelocation) -netdev user,id=network01 -device sungem,netdev=network01 -device VGA,edid=on -g 1366x768x32


Mac OS X DP2-------------


Start Setup-----------------------------


qemu-system-ppc -L pc-bios -m 256 -hda (harddiskimagelocation) -boot d -prom-env "boot-device=cd:9,\\:tbxi" -prom-env "auto-boot?=true" -M g3beige -cpu g3 -prom-env "boot-args=-v" -display sdl -device VGA,edid=on -g 1366x768x32 -cdrom (ISOlocation)

qemu-system-ppc -L pc-bios -m 256 -hda (harddiskimagelocation) -boot c -prom-env "boot-device=hd:9,\\:tbxi" -prom-env "auto-boot?=true" -M g3beige -cpu g3 -prom-env "boot-args=-v" -display sdl -device VGA,edid=on -g 1366x768x32
